struct conn_table* initTcpTable()
{
    struct conn_table* tcp_table;

    tcp_table->next_entry = 0;
    tcp_table->first = NULL;
    tcp_table->last = NULL;

    return tcp_table;
}

/*  
    Función de búsqueda de entradas en la tabla de conexiones.
    Recibe la tabla en la que buscar y la 5-tupla identificadora.
    Devuelve la posición del elemento si se encuentra en la tabla o -1 en caso contrario.
*/

int lookupEntry(struct conn_table *tcp_table, struct five_tuple id)
{
    int i = 0;

    while (i<tcp_table->next_entry && !sameFlow(tcp_table->connections[i].id, id))
        i++;

    return i < tcp_table->next_entry;
}

/*
    Función de comparación de flujos tcp.
    Recibe dos 5-tuplas y decide si pertenecen al mismo flujo.
    Devuelve 1 en ese caso y 0 en caso contrario.
*/
int sameFlow(struct five_tuple flow1, struct five_tuple flow2)
{
    int res = 0;
    
    if (flow1.src_ip == flow2.src_ip)
    {
        if (flow1.dst_ip == flow2.dst_ip &&
            flow1.src_port == flow2.src_port &&
            flow1.dst_port ==flow2.dst_port)

            res = flow1.proto == flow2.proto;
    }
    else if (flow1.src_ip == flow2.dst_ip &&
            flow1.dst_ip == flow2.src_ip &&
            flow1.src_port == flow2.dst_port &&
            flow1.dst_port == flow2.src_port)

            res = flow1.proto == flow2.proto;
    
    return res;
}
